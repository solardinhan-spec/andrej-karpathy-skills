## 5.3 API 클라이언트

```typescript
// src/lib/api.ts
import axios from 'axios';
import { auth } from './firebase';

const apiClient = axios.create({
  baseURL: process.env.NEXT_PUBLIC_API_URL || 'http://localhost:3000'
});

// 모든 요청에 Firebase ID 토큰 자동 첨부
apiClient.interceptors.request.use(async (config) => {
  const user = auth.currentUser;
  if (user) {
    const token = await user.getIdToken();
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

export default apiClient;
```

## 5.4 도면 대시보드 메인 페이지

```tsx
// src/app/drawings/page.tsx
'use client';

import { useState } from 'react';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import apiClient from '@/lib/api';
import StatusBadge from '@/components/StatusBadge';
import DrawingUpload from '@/components/DrawingUpload';
import ApprovalModal from '@/components/ApprovalModal';
import { Part } from '@/types';

async function fetchParts(): Promise<Part[]> {
  const res = await apiClient.get('/api/parts');
  return res.data.data;
}

export default function DrawingsPage() {
  const queryClient = useQueryClient();
  const [selectedPart, setSelectedPart] = useState<Part | null>(null);
  const [showUpload, setShowUpload] = useState(false);
  const [filterStatus, setFilterStatus] = useState<string>('');

  const { data: parts = [], isLoading } = useQuery({
    queryKey: ['parts', filterStatus],
    queryFn: () => fetchParts()
  });

  const approveMutation = useMutation({
    mutationFn: ({ partNo, revision }: { partNo: string; revision: string }) =>
      apiClient.post('/api/drawings/approve', { partNo, revision }),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['parts'] });
      setSelectedPart(null);
    }
  });

  const filteredParts = filterStatus
    ? parts.filter(p => p.status === filterStatus)
    : parts;

  return (
    <div className="min-h-screen bg-gray-50">
      {/* 헤더 */}
      <header className="bg-white border-b border-gray-200 px-6 py-4">
        <div className="max-w-7xl mx-auto flex items-center justify-between">
          <div>
            <h1 className="text-2xl font-bold text-gray-900">반도체 장비 PDM</h1>
            <p className="text-sm text-gray-500 mt-0.5">도면 관리 · 버전 이력 · 승인 워크플로우</p>
          </div>
          <button
            onClick={() => setShowUpload(true)}
            className="bg-blue-600 hover:bg-blue-700 text-white text-sm font-medium px-4 py-2 rounded-lg transition-colors"
          >
            + 도면 업로드
          </button>
        </div>
      </header>

      <main className="max-w-7xl mx-auto px-6 py-8">
        {/* 통계 카드 */}
        <div className="grid grid-cols-4 gap-4 mb-8">
          {[
            { label: '전체 품목', value: parts.length, color: 'blue' },
            { label: '작업 중', value: parts.filter(p => p.status === 'In Work').length, color: 'yellow' },
            { label: '배포 완료', value: parts.filter(p => p.status === 'Released').length, color: 'green' },
            { label: '폐기', value: parts.filter(p => p.status === 'Obsolete').length, color: 'gray' }
          ].map(({ label, value, color }) => (
            <div key={label} className="bg-white rounded-xl border border-gray-200 p-5">
              <p className="text-sm text-gray-500">{label}</p>
              <p className={`text-3xl font-bold text-${color}-600 mt-1`}>{value}</p>
            </div>
          ))}
        </div>

        {/* 필터 */}
        <div className="flex gap-2 mb-4">
          {['', 'In Work', 'Released', 'Obsolete'].map(status => (
            <button
              key={status}
              onClick={() => setFilterStatus(status)}
              className={`px-3 py-1.5 rounded-full text-sm font-medium transition-colors ${
                filterStatus === status
                  ? 'bg-blue-600 text-white'
                  : 'bg-white border border-gray-200 text-gray-600 hover:bg-gray-50'
              }`}
            >
              {status || '전체'}
            </button>
          ))}
        </div>

        {/* 품목 테이블 */}
        {isLoading ? (
          <div className="bg-white rounded-xl border border-gray-200 p-12 text-center text-gray-400">
            데이터를 불러오는 중...
          </div>
        ) : (
          <div className="bg-white rounded-xl border border-gray-200 overflow-hidden">
            <table className="w-full">
              <thead>
                <tr className="bg-gray-50 border-b border-gray-200">
                  {['품번', '품명', 'REV', '재질', '무게(kg)', '상태', '액션'].map(h => (
                    <th key={h} className="px-4 py-3 text-left text-xs font-semibold text-gray-500 uppercase tracking-wider">
                      {h}
                    </th>
                  ))}
                </tr>
              </thead>
              <tbody className="divide-y divide-gray-100">
                {filteredParts.map((part) => (
                  <tr key={part.partNo} className="hover:bg-gray-50 transition-colors">
                    <td className="px-4 py-3">
                      <a href={`/parts/${part.partNo}`} className="font-mono text-sm font-bold text-blue-600 hover:underline">
                        {part.partNo}
                      </a>
                    </td>
                    <td className="px-4 py-3 text-sm text-gray-900">{part.name}</td>
                    <td className="px-4 py-3 font-mono text-sm text-center text-gray-700">{part.revision}</td>
                    <td className="px-4 py-3 text-sm text-gray-500">{part.material}</td>
                    <td className="px-4 py-3 text-sm text-gray-500 text-right">{part.weight?.toFixed(2) || '-'}</td>
                    <td className="px-4 py-3">
                      <StatusBadge status={part.status} />
                    </td>
                    <td className="px-4 py-3">
                      {part.status === 'In Work' ? (
                        <button
                          onClick={() => setSelectedPart(part)}
                          className="text-xs bg-blue-50 hover:bg-blue-100 text-blue-700 font-medium px-3 py-1.5 rounded-lg transition-colors"
                        >
                          승인 요청
                        </button>
                      ) : part.status === 'Released' ? (
                        <span className="text-xs text-green-600 font-medium">✓ 배포 완료</span>
                      ) : (
                        <span className="text-xs text-gray-400">폐기됨</span>
                      )}
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>

            {filteredParts.length === 0 && (
              <div className="text-center py-12 text-gray-400 text-sm">
                조건에 맞는 품목이 없습니다.
              </div>
            )}
          </div>
        )}
      </main>

      {/* 승인 확인 모달 */}
      {selectedPart && (
        <ApprovalModal
          part={selectedPart}
          onConfirm={() => approveMutation.mutate({
            partNo: selectedPart.partNo,
            revision: selectedPart.revision
          })}
          onClose={() => setSelectedPart(null)}
          isLoading={approveMutation.isPending}
        />
      )}

      {/* 도면 업로드 패널 */}
      {showUpload && <DrawingUpload onClose={() => setShowUpload(false)} />}
    </div>
  );
}
```

// src/components/StatusBadge.tsx
type Status = 'In Work' | 'In Review' | 'Released' | 'Obsolete';

const statusConfig: Record<Status, { label: string; className: string }> = {
  'In Work':   { label: '작업 중',   className: 'bg-yellow-100 text-yellow-800 border-yellow-200' },
  'In Review': { label: '검토 중',   className: 'bg-purple-100 text-purple-800 border-purple-200' },
  'Released':  { label: '배포 완료', className: 'bg-green-100 text-green-800 border-green-200' },
  'Obsolete':  { label: '폐기',      className: 'bg-gray-100 text-gray-500 border-gray-200' }
};

export default function StatusBadge({ status }: { status: string }) {
  const config = statusConfig[status as Status] ?? { label: status, className: 'bg-gray-100 text-gray-600' };
  return (
    <span className={`inline-flex items-center px-2.5 py-0.5 rounded-full text-xs font-medium border ${config.className}`}>
      {config.label}
    </span>
  );
}

## 5.6 BOM 트리 뷰어 컴포넌트

```tsx
// src/components/BomTree.tsx
'use client';

import { useState } from 'react';
import { useQuery } from '@tanstack/react-query';
import apiClient from '@/lib/api';
import StatusBadge from './StatusBadge';

interface BomNode {
  partNo: string;
  name: string;
  revision: string;
  material: string;
  status: string;
  qty?: number;
  unit?: string;
  children?: BomNode[] | null;
}

function BomNodeRow({ node, level = 0 }: { node: BomNode; level?: number }) {
  const [expanded, setExpanded] = useState(level < 2);
  const hasChildren = node.children && node.children.length > 0;

  return (
    <>
      <tr className="hover:bg-gray-50 border-b border-gray-100">
        <td className="py-2 px-4">
          <div className="flex items-center gap-1" style={{ paddingLeft: `${level * 24}px` }}>
            {hasChildren ? (
              <button
                onClick={() => setExpanded(!expanded)}
                className="w-5 h-5 flex items-center justify-center text-gray-400 hover:text-gray-700"
              >
                {expanded ? '▼' : '▶'}
              </button>
            ) : (
              <span className="w-5 h-5" />
            )}
            <span className="font-mono text-sm text-blue-600">{node.partNo}</span>
          </div>
        </td>
        <td className="py-2 px-4 text-sm text-gray-700">{node.name}</td>
        <td className="py-2 px-4 text-sm font-mono text-center">{node.revision}</td>
        <td className="py-2 px-4 text-sm text-gray-500">{node.material}</td>
        <td className="py-2 px-4 text-sm text-center">{node.qty ?? '-'}</td>
        <td className="py-2 px-4 text-sm text-center text-gray-400">{node.unit ?? 'EA'}</td>
        <td className="py-2 px-4"><StatusBadge status={node.status} /></td>
      </tr>
      {expanded && hasChildren && node.children!.map((child, i) => (
        <BomNodeRow key={`${child.partNo}-${i}`} node={child} level={level + 1} />
      ))}
    </>
  );
}

export default function BomTree({ rootPartNo }: { rootPartNo: string }) {
  const { data, isLoading, error } = useQuery({
    queryKey: ['bom-tree', rootPartNo],
    queryFn: () => apiClient.get(`/api/bom/${rootPartNo}/tree`).then(r => r.data.bom)
  });

  if (isLoading) return <div className="text-center py-8 text-gray-400">BOM 구조 로딩 중...</div>;
  if (error) return <div className="text-center py-8 text-red-400">BOM 데이터를 불러올 수 없습니다.</div>;
  if (!data) return null;

  return (
    <div className="overflow-x-auto">
      <table className="w-full text-sm">
        <thead className="bg-gray-50">
          <tr>
            {['품번', '품명', 'REV', '재질', 'QTY', 'UNIT', '상태'].map(h => (
              <th key={h} className="px-4 py-2 text-left text-xs font-semibold text-gray-500 uppercase">
                {h}
              </th>
            ))}
          </tr>
        </thead>
        <tbody>
          <BomNodeRow node={data} />
        </tbody>
      </table>
    </div>
  );
}
```

## 5.7 도면 업로드 컴포넌트 (드래그 앤 드롭)

```tsx
// src/components/DrawingUpload.tsx
'use client';

import { useState, useCallback } from 'react';
import { useDropzone } from 'react-dropzone';
import { useMutation, useQueryClient } from '@tanstack/react-query';
import apiClient from '@/lib/api';

export default function DrawingUpload({ onClose }: { onClose: () => void }) {
  const queryClient = useQueryClient();
  const [partNo, setPartNo] = useState('');
  const [changeReason, setChangeReason] = useState('');
  const [pdfFile, setPdfFile] = useState<File | null>(null);
  const [cadFile, setCadFile] = useState<File | null>(null);

  const onDropPdf = useCallback((files: File[]) => setPdfFile(files[0] ?? null), []);
  const onDropCad = useCallback((files: File[]) => setCadFile(files[0] ?? null), []);

  const { getRootProps: getPdfProps, getInputProps: getPdfInput, isDragActive: isPdfDrag } =
    useDropzone({ onDrop: onDropPdf, accept: { 'application/pdf': ['.pdf'] }, maxFiles: 1 });

  const { getRootProps: getCadProps, getInputProps: getCadInput, isDragActive: isCadDrag } =
    useDropzone({
      onDrop: onDropCad,
      accept: { 'application/octet-stream': ['.step', '.stp', '.iges', '.dwg'] },
      maxFiles: 1
    });

  const uploadMutation = useMutation({
    mutationFn: async () => {
      if (!pdfFile || !partNo) throw new Error('품번과 PDF 파일은 필수입니다.');
      const formData = new FormData();
      formData.append('partNo', partNo);
      formData.append('changeReason', changeReason);
      formData.append('pdf', pdfFile);
      if (cadFile) formData.append('cad', cadFile);
      return apiClient.post('/api/drawings/upload', formData, {
        headers: { 'Content-Type': 'multipart/form-data' }
      });
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['parts'] });
      onClose();
    }
  });

  return (
    <div className="fixed inset-0 bg-black/50 flex items-center justify-center z-50">
      <div className="bg-white rounded-2xl shadow-2xl w-full max-w-lg mx-4 p-6">
        <h2 className="text-lg font-bold text-gray-900 mb-5">도면 업로드</h2>

        <div className="space-y-4">
          <div>
            <label className="block text-sm font-medium text-gray-700 mb-1">품번 *</label>
            <input
              type="text"
              value={partNo}
              onChange={e => setPartNo(e.target.value.toUpperCase())}
              placeholder="예: CH-001"
              className="w-full border border-gray-300 rounded-lg px-3 py-2 text-sm font-mono focus:ring-2 focus:ring-blue-500 focus:border-transparent"
            />
          </div>

          <div>
            <label className="block text-sm font-medium text-gray-700 mb-1">변경 사유</label>
            <input
              type="text"
              value={changeReason}
              onChange={e => setChangeReason(e.target.value)}
              placeholder="예: 재질 변경 SUS304 → SUS316L"
              className="w-full border border-gray-300 rounded-lg px-3 py-2 text-sm"
            />
          </div>

          {/* PDF 드롭존 */}
          <div>
            <label className="block text-sm font-medium text-gray-700 mb-1">PDF 도면 *</label>
            <div
              {...getPdfProps()}
              className={`border-2 border-dashed rounded-xl p-6 text-center cursor-pointer transition-colors ${
                isPdfDrag ? 'border-blue-500 bg-blue-50' : pdfFile ? 'border-green-500 bg-green-50' : 'border-gray-300 hover:border-gray-400'
              }`}
            >
              <input {...getPdfInput()} />
              {pdfFile ? (
                <p className="text-sm text-green-700 font-medium">{pdfFile.name}</p>
              ) : (
                <p className="text-sm text-gray-400">PDF 파일을 드래그하거나 클릭하여 선택</p>
              )}
            </div>
          </div>

          {/* CAD 드롭존 */}
          <div>
            <label className="block text-sm font-medium text-gray-700 mb-1">CAD 파일 (STEP/IGES 선택사항)</label>
            <div
              {...getCadProps()}
              className={`border-2 border-dashed rounded-xl p-4 text-center cursor-pointer transition-colors ${
                isCadDrag ? 'border-purple-500 bg-purple-50' : cadFile ? 'border-green-500 bg-green-50' : 'border-gray-300 hover:border-gray-400'
              }`}
            >
              <input {...getCadInput()} />
              {cadFile ? (
                <p className="text-sm text-green-700 font-medium">{cadFile.name}</p>
              ) : (
                <p className="text-sm text-gray-400">.step, .stp, .iges, .dwg</p>
              )}
            </div>
          </div>
        </div>

        {uploadMutation.isError && (
          <p className="mt-3 text-sm text-red-600">
            {(uploadMutation.error as Error).message}
          </p>
        )}

        <div className="flex gap-3 mt-6">
          <button onClick={onClose} className="flex-1 border border-gray-300 text-gray-700 text-sm font-medium py-2.5 rounded-xl hover:bg-gray-50">
            취소
          </button>
          <button
            onClick={() => uploadMutation.mutate()}
            disabled={!pdfFile || !partNo || uploadMutation.isPending}
            className="flex-1 bg-blue-600 hover:bg-blue-700 disabled:bg-gray-300 text-white text-sm font-medium py-2.5 rounded-xl transition-colors"
          >
            {uploadMutation.isPending ? '업로드 중...' : '업로드'}
          </button>
        </div>
      </div>
    </div>
  );
}
```
