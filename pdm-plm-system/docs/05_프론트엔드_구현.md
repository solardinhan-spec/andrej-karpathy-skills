```tsx
// src/app/login/page.tsx
'use client';
import { useState } from 'react';
import { signInWithEmailAndPassword } from 'firebase/auth';
import { auth } from '@/lib/firebase';
import { useRouter } from 'next/navigation';

export default function LoginPage() {
  const router = useRouter();
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');

          placeholder="이메일" className="w-full border rounded-lg px-3 py-2 mb-3 text-sm" />
```

## 5.5 APS DWG 웹 뷰어 컴포넌트

```tsx
// src/components/ApsViewer.tsx
'use client';
import { useEffect, useRef } from 'react';
import apiClient from '@/lib/api';

// Autodesk Viewer 스크립트/스타일 CDN
const VIEWER_JS = 'https://developer.api.autodesk.com/modelderivative/v2/viewers/7.*/viewer3D.min.js';
const VIEWER_CSS = 'https://developer.api.autodesk.com/modelderivative/v2/viewers/7.*/style.min.css';

function loadScript(src: string) {
  return new Promise<void>((resolve, reject) => {
    if (document.querySelector(`script[src="${src}"]`)) return resolve();
    const s = document.createElement('script');
    s.src = src; s.onload = () => resolve(); s.onerror = reject;
    document.body.appendChild(s);
  });
}

export default function ApsViewer({ dwgNo }: { dwgNo: string }) {
  const containerRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    let viewer: any;
    (async () => {
      // CSS 로드
      if (!document.querySelector(`link[href="${VIEWER_CSS}"]`)) {
        const link = document.createElement('link');
        link.rel = 'stylesheet'; link.href = VIEWER_CSS;
        document.head.appendChild(link);
      }
      await loadScript(VIEWER_JS);

      // 뷰어 정보 + APS 토큰
      const [{ data: info }, { data: tokenRes }] = await Promise.all([
        apiClient.get(`/api/search/${dwgNo}/viewer`),
        apiClient.get('/api/search/aps/token')
      ]);
      if (!info.apsUrn) return;
      if (info.apsStatus !== 'complete') {
        containerRef.current!.innerHTML =
          `<div style="padding:2rem;color:#888">DWG 변환 중입니다 (상태: ${info.apsStatus}). 잠시 후 새로고침하세요.</div>`;
        return;
      }

      const Autodesk = (window as any).Autodesk;
      Autodesk.Viewing.Initializer({
        env: 'AutodeskProduction',
        getAccessToken: (cb: any) => cb(tokenRes.access_token, tokenRes.expires_in)
      }, () => {
        viewer = new Autodesk.Viewing.GuiViewer3D(containerRef.current);
        viewer.start();
        Autodesk.Viewing.Document.load(
          `urn:${info.apsUrn}`,
          (doc: any) => viewer.loadDocumentNode(doc, doc.getRoot().getDefaultGeometry()),
          (err: any) => console.error('APS 로드 실패', err)
        );
      });
    })();

    return () => { if (viewer) viewer.finish(); };
  }, [dwgNo]);

  return <div ref={containerRef} className="w-full h-[600px] bg-gray-900 rounded-xl relative" />;
}
```

## 5.7 검토자 2명 승인 패널

```tsx
// src/components/ApprovalPanel.tsx
'use client';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import apiClient from '@/lib/api';
import { useAuth } from '@/hooks/useAuth';
import StatusBadge from './StatusBadge';

export default function ApprovalPanel({ dwgNo }: { dwgNo: string }) {
  const qc = useQueryClient();
  const { user } = useAuth();

  const { data } = useQuery({
    queryKey: ['drawing', dwgNo],
    queryFn: () => apiClient.get(`/api/drawings/${dwgNo}/history`).then(r => r.data.head)
  });

  const submitMut = useMutation({
    mutationFn: () => apiClient.post(`/api/drawings/${dwgNo}/submit`),
    onSuccess: () => qc.invalidateQueries({ queryKey: ['drawing', dwgNo] })
  });
  const approveMut = useMutation({
    mutationFn: () => apiClient.post(`/api/drawings/${dwgNo}/approve`),
    onSuccess: () => qc.invalidateQueries({ queryKey: ['drawing', dwgNo] })
  });

  if (!data) return null;
  const isAuthor = data.authors?.includes(user?.uid);
  const isReviewer = data.reviewers?.includes(user?.uid);
  const approvedCount = Object.values(data.approvals || {}).filter((a: any) => a.approved).length;
  const myApproved = data.approvals?.[user?.uid || '']?.approved;

  return (
    <div className="bg-white rounded-xl border p-5">
      <div className="flex items-center justify-between mb-4">
        <h2 className="font-semibold text-gray-900">승인 워크플로우</h2>
        <StatusBadge status={data.status} />
      </div>

      <div className="text-sm text-gray-600 space-y-1 mb-4">
        <p>담당자: {data.authors?.length}명</p>
        <p>검토자 승인: <span className="font-bold text-blue-600">{approvedCount} / 2</span></p>
      </div>

      {/* 담당자: In Work일 때 제출 */}
      {isAuthor && data.status === 'In Work' && (
        <button onClick={() => submitMut.mutate()} disabled={submitMut.isPending}
          className="w-full bg-yellow-500 hover:bg-yellow-600 text-white text-sm font-medium py-2.5 rounded-lg">
          검토 요청 (제출)
        </button>
      )}

      {/* 검토자: In Review이고 본인 미승인일 때 */}
      {isReviewer && data.status === 'In Review' && !myApproved && (
        <button onClick={() => approveMut.mutate()} disabled={approveMut.isPending}
          className="w-full bg-blue-600 hover:bg-blue-700 text-white text-sm font-medium py-2.5 rounded-lg">
          승인 ({approvedCount}/2 → {approvedCount + 1}/2)
        </button>
      )}

      {isReviewer && myApproved && data.status === 'In Review' && (
        <p className="text-sm text-green-600 text-center">✓ 본인 승인 완료. 다른 검토자 대기 중</p>
      )}
      {data.status === 'Released' && (
        <p className="text-sm text-green-600 text-center font-medium">✓ 검토자 2명 전원 승인 — 배포 완료</p>
      )}
    </div>
  );
}
```

## 5.9 DWG 단일 업로드

```tsx
// src/components/DrawingUpload.tsx
'use client';
import { useState, useCallback } from 'react';
import { useDropzone } from 'react-dropzone';
import { useMutation, useQueryClient } from '@tanstack/react-query';
import apiClient from '@/lib/api';

export default function DrawingUpload({ onClose }: { onClose: () => void }) {
  const qc = useQueryClient();
  const [itemCode, setItemCode] = useState('');
  const [dwgNo, setDwgNo] = useState('');     // 비우면 신규 도번 발급, 입력 시 Rev↑
  const [title, setTitle] = useState('');
  const [changeReason, setChangeReason] = useState('');
  const [authors, setAuthors] = useState('');     // 콤마 구분 uid
  const [reviewers, setReviewers] = useState(''); // 정확히 2명 uid
  const [dwg, setDwg] = useState<File | null>(null);

  const onDrop = useCallback((files: File[]) => setDwg(files[0] ?? null), []);
  const { getRootProps, getInputProps, isDragActive } = useDropzone({
    onDrop, accept: { 'application/acad': ['.dwg'] }, maxFiles: 1
  });

  const uploadMut = useMutation({
    mutationFn: async () => {
      const reviewerList = reviewers.split(',').map(s => s.trim()).filter(Boolean);
      if (reviewerList.length !== 2) throw new Error('검토자는 정확히 2명이어야 합니다.');
      if (!dwg || !itemCode || !title) throw new Error('품목코드 · 도면명 · DWG는 필수입니다.');

      const fd = new FormData();
      fd.append('itemCode', itemCode);
      if (dwgNo) fd.append('dwgNo', dwgNo);
      fd.append('title', title);
      fd.append('changeReason', changeReason);
      fd.append('authors', JSON.stringify(authors.split(',').map(s => s.trim()).filter(Boolean)));
      fd.append('reviewers', JSON.stringify(reviewerList));
      fd.append('dwg', dwg);
      return apiClient.post('/api/drawings/upload', fd, {
        headers: { 'Content-Type': 'multipart/form-data' }
      });
    },
    onSuccess: () => { qc.invalidateQueries({ queryKey: ['search'] }); onClose(); }
  });

  return (
    <div className="fixed inset-0 bg-black/50 flex items-center justify-center z-50">
      <div className="bg-white rounded-2xl shadow-2xl w-full max-w-lg mx-4 p-6 space-y-3">
        <h2 className="text-lg font-bold text-gray-900">DWG 도면 업로드</h2>

        <input value={itemCode} onChange={e => setItemCode(e.target.value.toUpperCase())}
          placeholder="품목코드 (기존 제품이면 동일 코드 입력 — 중복 허용)"
          className="w-full border rounded-lg px-3 py-2 text-sm font-mono" />
        <input value={dwgNo} onChange={e => setDwgNo(e.target.value.toUpperCase())}
          placeholder="도번 (비우면 신규 발급 / 입력 시 해당 도번 Revision↑)"
          className="w-full border rounded-lg px-3 py-2 text-sm font-mono" />
        <input value={title} onChange={e => setTitle(e.target.value)}
          placeholder="도면명 (예: 조립도)" className="w-full border rounded-lg px-3 py-2 text-sm" />
        <input value={changeReason} onChange={e => setChangeReason(e.target.value)}
          placeholder="변경 사유" className="w-full border rounded-lg px-3 py-2 text-sm" />
        <input value={authors} onChange={e => setAuthors(e.target.value)}
          placeholder="담당자 uid (콤마 구분, 여러 명)" className="w-full border rounded-lg px-3 py-2 text-sm" />
        <input value={reviewers} onChange={e => setReviewers(e.target.value)}
          placeholder="검토자 uid (정확히 2명, 콤마 구분)" className="w-full border rounded-lg px-3 py-2 text-sm" />

        <div {...getRootProps()}
          className={`border-2 border-dashed rounded-xl p-6 text-center cursor-pointer ${
            isDragActive ? 'border-blue-500 bg-blue-50' : dwg ? 'border-green-500 bg-green-50' : 'border-gray-300'
          }`}>
          <input {...getInputProps()} />
          <p className="text-sm text-gray-500">{dwg ? dwg.name : 'DWG 파일을 드래그하거나 클릭'}</p>
        </div>

        {uploadMut.isError && <p className="text-sm text-red-600">{(uploadMut.error as Error).message}</p>}

        <div className="flex gap-3 pt-2">
          <button onClick={onClose} className="flex-1 border text-gray-700 text-sm py-2.5 rounded-xl">취소</button>
          <button onClick={() => uploadMut.mutate()} disabled={uploadMut.isPending}
            className="flex-1 bg-blue-600 hover:bg-blue-700 disabled:bg-gray-300 text-white text-sm py-2.5 rounded-xl">
            {uploadMut.isPending ? '업로드 중...' : '업로드'}
          </button>
        </div>
      </div>
    </div>
  );
}
```

```tsx
          className="col-span-2 bg-blue-600 hover:bg-blue-700 text-white text-sm font-medium py-2.5 rounded-lg">
          계정 1회 발급
        </button>
      </div>

      {resetLink && (
        <div className="bg-green-50 border border-green-200 rounded-lg p-4 mb-6 text-sm">
          비밀번호 설정 링크를 사용자에게 전달하세요:
          <code className="block mt-1 break-all text-green-800">{resetLink}</code>
        </div>
      )}

      <table className="w-full text-sm bg-white rounded-xl border overflow-hidden">
        <thead className="bg-gray-50"><tr>
          {['이름', '이메일', '부서', '역할'].map(h => (
            <th key={h} className="px-4 py-2 text-left text-xs text-gray-500 uppercase">{h}</th>
          ))}
        </tr></thead>
        <tbody className="divide-y">
          {users.map((u: any) => (
            <tr key={u.uid}>
              <td className="px-4 py-2">{u.name}</td>
              <td className="px-4 py-2">{u.email}</td>
              <td className="px-4 py-2">{u.department}</td>
              <td className="px-4 py-2 font-mono">{u.role}</td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  );
}
```
