## 3.2 server.js (앱 진입점)

```javascript
// src/server.js
const express = require('express');
const cors = require('cors');
const helmet = require('helmet');
const rateLimit = require('express-rate-limit');
const admin = require('firebase-admin');
require('dotenv').config();

const partsRouter = require('./routes/parts');
const drawingsRouter = require('./routes/drawings');
const bomRouter = require('./routes/bom');
const ecnRouter = require('./routes/ecn');
const { verifyToken } = require('./middleware/auth');

// Firebase Admin 초기화 (Secret Manager에서 키 로드)
const initFirebase = () => {
  if (process.env.NODE_ENV === 'production') {
    // Cloud Run 환경: 서비스 계정 자동 인식
    admin.initializeApp({
      storageBucket: `${process.env.PROJECT_ID}.appspot.com`
    });
  } else {
    // 로컬 개발 환경
    const serviceAccount = require('../serviceAccountKey.json');
    admin.initializeApp({
      credential: admin.credential.cert(serviceAccount),
      storageBucket: process.env.STORAGE_BUCKET
    });
  }
};

initFirebase();

const app = express();

// 보안 헤더
app.use(helmet());

// CORS 설정 (운영 도메인만 허용)
app.use(cors({
  origin: process.env.NODE_ENV === 'production'
    ? ['https://pdm.company.com', 'https://pdm-app.web.app']
    : 'http://localhost:3001',
  credentials: true
}));

app.use(express.json({ limit: '10mb' }));

// API Rate Limiting (DoS 방어)
const apiLimiter = rateLimit({
  windowMs: 15 * 60 * 1000, // 15분
  max: 100,
  message: { error: '요청 횟수 초과. 잠시 후 다시 시도하세요.' }
});
app.use('/api/', apiLimiter);

// 헬스체크 (Cloud Run 헬스프로브)
app.get('/health', (req, res) => res.json({ status: 'ok', timestamp: new Date().toISOString() }));

// 인증이 필요한 API 라우터
app.use('/api/parts', verifyToken, partsRouter);
app.use('/api/drawings', verifyToken, drawingsRouter);
app.use('/api/bom', verifyToken, bomRouter);
app.use('/api/ecn', verifyToken, ecnRouter);

// 전역 에러 핸들러
app.use((err, req, res, next) => {
  console.error(`[ERROR] ${err.message}`, err.stack);
  res.status(err.status || 500).json({
    success: false,
    error: process.env.NODE_ENV === 'production' ? '서버 오류가 발생했습니다.' : err.message
  });
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => console.log(`PDM 서버 가동 중: 포트 ${PORT}`));
```

## 3.3 자동 채번 서비스

```javascript
// src/services/numbering.js
const admin = require('firebase-admin');

const db = admin.firestore();

/**
 * Firestore 트랜잭션 기반 동시성 안전 채번
 * @param {string} prefix - 품목 분류 코드 (CH, WC, AS, EL 등)
 * @param {number} padLength - 숫자 자릿수 (기본 3자리: 001)
 * @returns {string} - 예: "CH-001"
 */
async function generatePartNo(prefix, padLength = 3) {
  const sequenceRef = db.collection('sequences').doc(prefix);

  return db.runTransaction(async (transaction) => {
    const seqDoc = await transaction.get(sequenceRef);
    const nextSeq = seqDoc.exists ? seqDoc.data().currentSeq + 1 : 1;

    transaction.set(sequenceRef, {
      currentSeq: nextSeq,
      prefix,
      lastUpdated: admin.firestore.FieldValue.serverTimestamp()
    });

    return `${prefix}-${String(nextSeq).padStart(padLength, '0')}`;
  });
}

module.exports = { generatePartNo };
```

## 3.4 Revision 관리 서비스

```javascript
// src/services/revision.js

/**
 * 알파벳 리비전 자동 증가
 * A → B → ... → Z → AA → AB → ...
 */
function getNextRevision(currentRev) {
  if (!currentRev) return 'A';

  const chars = currentRev.split('');
  let carry = true;

  for (let i = chars.length - 1; i >= 0 && carry; i--) {
    const code = chars[i].charCodeAt(0);
    if (code < 90) { // 'Z' = 90
      chars[i] = String.fromCharCode(code + 1);
      carry = false;
    } else {
      chars[i] = 'A'; // Z → A + carry
    }
  }

  if (carry) chars.unshift('A'); // AA, AAA 확장

  return chars.join('');
}

/**
 * 리비전 비교 (정렬용)
 * A < B < ... < Z < AA < AB
 */
function compareRevisions(revA, revB) {
  if (revA.length !== revB.length) return revA.length - revB.length;
  return revA.localeCompare(revB);
}

module.exports = { getNextRevision, compareRevisions };
```

## 3.5 Parts API 라우터

```javascript
// src/routes/parts.js
const express = require('express');
const router = express.Router();
const admin = require('firebase-admin');
const Joi = require('joi');
const { generatePartNo } = require('../services/numbering');

const db = admin.firestore();

// 입력값 검증 스키마
const partSchema = Joi.object({
  prefix: Joi.string().uppercase().min(2).max(4).required(),
  name: Joi.string().min(2).max(100).required(),
  material: Joi.string().default('AL6061'),
  weight: Joi.number().min(0).default(0),
  description: Joi.string().max(500).optional()
});

// Part 등록
router.post('/', async (req, res, next) => {
  try {
    const { error, value } = partSchema.validate(req.body);
    if (error) return res.status(400).json({ success: false, error: error.details[0].message });

    const { prefix, name, material, weight, description } = value;

    const partNo = await generatePartNo(prefix);

    const newPart = {
      partNo,
      name,
      revision: 'A',
      material,
      weight,
      description: description || '',
      status: 'In Work',
      createdBy: req.user.uid,
      createdByName: req.user.name || req.user.email,
      createdAt: admin.firestore.FieldValue.serverTimestamp(),
      updatedAt: admin.firestore.FieldValue.serverTimestamp()
    };

    await db.collection('parts').doc(partNo).set(newPart);

    res.status(201).json({ success: true, partNo, data: newPart });
  } catch (err) {
    next(err);
  }
});

// Part 목록 조회 (페이징 지원)
router.get('/', async (req, res, next) => {
  try {
    const { status, prefix, limit = 50, startAfter } = req.query;

    let query = db.collection('parts').orderBy('createdAt', 'desc').limit(Number(limit));

    if (status) query = query.where('status', '==', status);
    if (prefix) query = query.where('partNo', '>=', prefix).where('partNo', '<', prefix + '');
    if (startAfter) {
      const lastDoc = await db.collection('parts').doc(startAfter).get();
      if (lastDoc.exists) query = query.startAfter(lastDoc);
    }

    const snapshot = await query.get();
    const parts = snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() }));

    res.json({ success: true, data: parts, count: parts.length });
  } catch (err) {
    next(err);
  }
});

// 특정 Part 조회
router.get('/:partNo', async (req, res, next) => {
  try {
    const doc = await db.collection('parts').doc(req.params.partNo).get();
    if (!doc.exists) return res.status(404).json({ success: false, error: '품번을 찾을 수 없습니다.' });

    res.json({ success: true, data: { id: doc.id, ...doc.data() } });
  } catch (err) {
    next(err);
  }
});

// Part 도면 이력 조회
router.get('/:partNo/history', async (req, res, next) => {
  try {
    const snapshot = await db.collection('drawings')
      .where('partNo', '==', req.params.partNo)
      .orderBy('uploadedAt', 'desc')
      .get();

    const history = snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() }));
    res.json({ success: true, data: history });
  } catch (err) {
    next(err);
  }
});

module.exports = router;
```

## 3.6 Drawings API 라우터 (파일 업로드 + Revision 관리)

```javascript
// src/routes/drawings.js
const express = require('express');
const router = express.Router();
const admin = require('firebase-admin');
const multer = require('multer');
const path = require('path');
const { getNextRevision } = require('../services/revision');

const db = admin.firestore();
const bucket = admin.storage().bucket();

// 허용 파일 형식 검증
const fileFilter = (req, file, cb) => {
  const allowedPdf = ['.pdf'];
  const allowedCad = ['.step', '.stp', '.iges', '.igs', '.dwg', '.dxf'];
  const ext = path.extname(file.originalname).toLowerCase();

  if (file.fieldname === 'pdf' && allowedPdf.includes(ext)) return cb(null, true);
  if (file.fieldname === 'cad' && allowedCad.includes(ext)) return cb(null, true);

  cb(new Error(`허용되지 않는 파일 형식: ${ext}`));
};

const upload = multer({
  storage: multer.memoryStorage(),
  limits: { fileSize: 100 * 1024 * 1024 }, // 100MB 제한
  fileFilter
});

// 도면 업로드 + Revision 자동 관리
router.post('/upload', upload.fields([{ name: 'pdf', maxCount: 1 }, { name: 'cad', maxCount: 1 }]), async (req, res, next) => {
  try {
    const { partNo, changeReason } = req.body;

    if (!req.files?.pdf) {
      return res.status(400).json({ success: false, error: 'PDF 도면 파일이 필요합니다.' });
    }

    // Part 존재 여부 확인
    const partRef = db.collection('parts').doc(partNo);
    const partDoc = await partRef.get();
    if (!partDoc.exists) {
      return res.status(404).json({ success: false, error: `품번 ${partNo}이 존재하지 않습니다.` });
    }

    const partData = partDoc.data();

    // Revision 결정: Released 상태면 다음 Rev, 아니면 현재 Rev 유지
    const nextRev = partData.status === 'Released'
      ? getNextRevision(partData.revision)
      : partData.revision;

    const drawingId = `${partNo}-${nextRev}`;
    const timestamp = Date.now();

    // Cloud Storage 업로드
    const pdfFile = req.files.pdf[0];
    const pdfPath = `drawings/${partNo}/REV_${nextRev}/${partNo}_REV_${nextRev}_${timestamp}.pdf`;
    const pdfBlob = bucket.file(pdfPath);

    await pdfBlob.save(pdfFile.buffer, {
      contentType: 'application/pdf',
      metadata: {
        partNo,
        revision: nextRev,
        uploadedBy: req.user.uid
      }
    });

    // Signed URL 생성 (7일 유효)
    const [pdfSignedUrl] = await pdfBlob.getSignedUrl({
      action: 'read',
      expires: Date.now() + 7 * 24 * 60 * 60 * 1000
    });

    let cadPath = null;
    if (req.files?.cad) {
      const cadFile = req.files.cad[0];
      const cadExt = path.extname(cadFile.originalname).toLowerCase();
      cadPath = `drawings/${partNo}/REV_${nextRev}/${partNo}_REV_${nextRev}${cadExt}`;
      const cadBlob = bucket.file(cadPath);
      await cadBlob.save(cadFile.buffer, { contentType: 'application/octet-stream' });
    }

    // Firestore 배치 쓰기 (원자적 업데이트)
    const batch = db.batch();

    batch.set(db.collection('drawings').doc(drawingId), {
      partNo,
      drawingNo: `DWG-${partNo}`,
      revision: nextRev,
      filePath: pdfPath,
      fileUrl: `gs://${bucket.name}/${pdfPath}`,
      cadPath,
      status: 'In Work',
      changeReason: changeReason || '',
      uploadedBy: req.user.uid,
      uploadedByName: req.user.name || req.user.email,
      uploadedAt: admin.firestore.FieldValue.serverTimestamp()
    });

    batch.update(partRef, {
      revision: nextRev,
      status: 'In Work',
      updatedAt: admin.firestore.FieldValue.serverTimestamp()
    });

    await batch.commit();

    res.json({
      success: true,
      drawingId,
      partNo,
      revision: nextRev,
      viewUrl: pdfSignedUrl
    });
  } catch (err) {
    next(err);
  }
});

// 도면 승인 (Released 상태로 전환)
router.post('/approve', async (req, res, next) => {
  try {
    const { partNo, revision } = req.body;

    // 승인 권한 확인 (reviewer 또는 admin만 가능)
    if (!['reviewer', 'admin'].includes(req.user.role)) {
      return res.status(403).json({ success: false, error: '승인 권한이 없습니다.' });
    }

    const drawingId = `${partNo}-${revision}`;
    const drawingRef = db.collection('drawings').doc(drawingId);
    const partRef = db.collection('parts').doc(partNo);

    const drawingDoc = await drawingRef.get();
    if (!drawingDoc.exists) {
      return res.status(404).json({ success: false, error: `도면 ${drawingId}을 찾을 수 없습니다.` });
    }

    if (drawingDoc.data().status === 'Released') {
      return res.status(400).json({ success: false, error: '이미 배포된 도면입니다.' });
    }

    const batch = db.batch();

    // 기존 Released 도면을 Obsolete로 전환
    const oldReleasedSnapshot = await db.collection('drawings')
      .where('partNo', '==', partNo)
      .where('status', '==', 'Released')
      .get();

    oldReleasedSnapshot.forEach(doc => {
      batch.update(doc.ref, {
        status: 'Obsolete',
        obsoletedAt: admin.firestore.FieldValue.serverTimestamp()
      });
    });

    // 현재 도면 승인 처리
    batch.update(drawingRef, {
      status: 'Released',
      approvedBy: req.user.uid,
      approvedByName: req.user.name || req.user.email,
      approvedDate: admin.firestore.FieldValue.serverTimestamp()
    });

    // Part 마스터 동기화
    batch.update(partRef, {
      status: 'Released',
      updatedAt: admin.firestore.FieldValue.serverTimestamp()
    });

    await batch.commit();

    res.json({
      success: true,
      message: `${drawingId} 도면이 배포(Released) 완료되었습니다.`
    });
  } catch (err) {
    next(err);
  }
});

// 도면 다운로드 Signed URL 발급
router.get('/:drawingId/download', async (req, res, next) => {
  try {
    const doc = await db.collection('drawings').doc(req.params.drawingId).get();
    if (!doc.exists) return res.status(404).json({ success: false, error: '도면을 찾을 수 없습니다.' });

    const { filePath } = doc.data();
    const file = bucket.file(filePath);

    const [url] = await file.getSignedUrl({
      action: 'read',
      expires: Date.now() + 60 * 60 * 1000 // 1시간 유효
    });

    res.json({ success: true, downloadUrl: url, expiresIn: '1시간' });
  } catch (err) {
    next(err);
  }
});

module.exports = router;
```

## 3.7 Firebase Auth 미들웨어

```javascript
// src/middleware/auth.js
const admin = require('firebase-admin');

async function verifyToken(req, res, next) {
  const authHeader = req.headers.authorization;

  if (!authHeader?.startsWith('Bearer ')) {
    return res.status(401).json({ success: false, error: '인증 토큰이 필요합니다.' });
  }

  const token = authHeader.split('Bearer ')[1];

  try {
    const decoded = await admin.auth().verifyIdToken(token);

    // Firestore에서 사용자 역할 조회
    const userDoc = await admin.firestore().collection('users').doc(decoded.uid).get();
    const role = userDoc.exists ? userDoc.data().role : 'viewer';

    req.user = {
      uid: decoded.uid,
      email: decoded.email,
      name: decoded.name,
      role
    };

    next();
  } catch (err) {
    res.status(401).json({ success: false, error: '유효하지 않은 토큰입니다.' });
  }
}

module.exports = { verifyToken };
```
