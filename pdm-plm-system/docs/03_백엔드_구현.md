# 03. 백엔드 구현 (Express on Cloud Functions 2nd gen + Firestore)

> **배포 형태:** Docker 미사용. 기존 Express 앱을 그대로 작성하되,
> 진입점만 Cloud Functions 2nd gen `onRequest(app)`로 래핑하여 서버리스로 배포한다.

## 3.1 프로젝트 초기화

> 본 저장소에서는 이 함수 소스 디렉토리가 `backend/`이다 (firebase.json의
> `functions.source`). 아래 예시의 `pdm-functions`는 동일한 디렉토리를 가리킨다.

```bash
mkdir pdm-functions && cd pdm-functions

  @google-cloud/storage @google-cloud/secret-manager googleapis \
  multer joi forge-apis
npm install -D firebase-functions-test eslint jest
```

## 3.3 자동 채번 서비스 (itemCode + dwgNo)

```javascript
// src/services/numbering.js
const admin = require('firebase-admin');
const db = admin.firestore();

/**
 * 트랜잭션 기반 단조 증가 시퀀스
 */
async function nextSequence(seqKey) {
  const ref = db.collection('sequences').doc(seqKey);
  return db.runTransaction(async (tx) => {
    const doc = await tx.get(ref);
    const next = doc.exists ? doc.data().currentSeq + 1 : 1;
    tx.set(ref, { currentSeq: next, updatedAt: admin.firestore.FieldValue.serverTimestamp() });
    return next;
  });
}

/**
 * 품목코드 채번: 분류 Prefix + 3자리 (예: CH-001)
 * ※ 동일 제품이면 기존 itemCode를 재사용하므로 신규 제품일 때만 호출
 */
async function generateItemCode(prefix) {
  const seq = await nextSequence(`ITEM_${prefix}`);
  return `${prefix}-${String(seq).padStart(3, '0')}`;
}

/**
 * 도번 채번: 전역 유일 (예: DWG-0001)
 */
async function generateDwgNo() {
  const seq = await nextSequence('DWG');
  return `DWG-${String(seq).padStart(4, '0')}`;
}

module.exports = { generateItemCode, generateDwgNo };
```

## 3.4 Revision 관리 서비스

```javascript
// src/services/revision.js
// A → B → ... → Z → AA → AB ... (dwgNo별 독립 증가)
function getNextRevision(currentRev) {
  if (!currentRev) return 'A';
  const chars = currentRev.split('');
  let carry = true;
  for (let i = chars.length - 1; i >= 0 && carry; i--) {
    const code = chars[i].charCodeAt(0);
    if (code < 90) { chars[i] = String.fromCharCode(code + 1); carry = false; }
    else { chars[i] = 'A'; }
  }
  if (carry) chars.unshift('A');
  return chars.join('');
}

module.exports = { getNextRevision };
```

## 3.5 APS(Forge) DWG 변환 서비스

```javascript
// src/services/aps.js
const { AuthClientTwoLegged, ObjectsApi, DerivativesApi } = require('forge-apis');

const APS_BUCKET = process.env.APS_BUCKET_KEY || 'pdm-dwg-derivatives';

async function getApsToken() {
  const client = new AuthClientTwoLegged(
    process.env.APS_CLIENT_ID,
    process.env.APS_CLIENT_SECRET,
    ['data:read', 'data:write', 'data:create', 'bucket:read', 'bucket:create'],
    true
  );
  const cred = await client.authenticate();
  return cred;
}

/**
 * DWG 버퍼를 APS에 업로드하고 변환 잡을 등록 → URN 반환
 */
async function translateDwg(dwgNo, revision, buffer) {
  const cred = await getApsToken();
  const objectKey = `${dwgNo}_REV_${revision}.dwg`;

  // 1) OSS 업로드 (버킷이 없으면 최초 1회 생성 필요 — 별도 부트스트랩)
  await new ObjectsApi().uploadResources(
    APS_BUCKET,
    [{ objectKey, data: buffer }],
    {},
    null,
    cred
  );

  // 2) URN 생성 (base64url)
  const urn = Buffer.from(`urn:adsk.objects:os.object:${APS_BUCKET}/${objectKey}`)
    .toString('base64')
    .replace(/=/g, '')
    .replace(/\+/g, '-')
    .replace(/\//g, '_');

  // 3) 변환 잡 등록 (DWG → SVF2)
  await new DerivativesApi().translate(
    {
      input: { urn },
      output: { formats: [{ type: 'svf2', views: ['2d', '3d'] }] }
    },
    {},
    null,
    cred
  );

  return urn;
}

/**
 * 변환 상태 조회 (pending | inprogress | complete | failed)
 */
async function getTranslationStatus(urn) {
  const cred = await getApsToken();
  const manifest = await new DerivativesApi().getManifest(urn, {}, null, cred);
  return manifest.body.status;
}

module.exports = { translateDwg, getTranslationStatus, getApsToken };
```

## 3.6 Items API (품목 마스터)

```javascript
// src/routes/items.js
const express = require('express');
const router = express.Router();
const admin = require('firebase-admin');
const Joi = require('joi');
const { generateItemCode } = require('../services/numbering');

const db = admin.firestore();

const itemSchema = Joi.object({
  prefix: Joi.string().uppercase().min(2).max(4).required(), // 신규 제품일 때
  itemCode: Joi.string().optional(),  // 기존 제품에 도면 추가 시 직접 지정(중복 허용)
  name: Joi.string().min(2).max(100).required(),
  category: Joi.string().max(50).default(''),
  material: Joi.string().default('AL6061'),
  weight: Joi.number().min(0).default(0)
});

// 품목 등록 (신규 제품) — 기존 제품 재사용 시 itemCode를 직접 전달
router.post('/', async (req, res, next) => {
  try {
    if (!['designer', 'admin'].includes(req.user.role)) {
      return res.status(403).json({ success: false, error: '품목 등록 권한이 없습니다.' });
    }
    const { error, value } = itemSchema.validate(req.body);
    if (error) return res.status(400).json({ success: false, error: error.details[0].message });

    // itemCode가 주어지면 기존 제품(중복 허용), 없으면 신규 채번
    const itemCode = value.itemCode || await generateItemCode(value.prefix);
    const ref = db.collection('items').doc(itemCode);
    const existing = await ref.get();

    if (existing.exists) {
      // 기존 제품 — 새로 만들지 않고 그대로 사용 (중복 itemCode 허용 정책)
      return res.json({ success: true, itemCode, reused: true, data: existing.data() });
    }

    const newItem = {
      itemCode,
      name: value.name,
      category: value.category,
      material: value.material,
      weight: value.weight,
      latestRev: null,
      drawingCount: 0,
      createdBy: req.user.uid,
      createdAt: admin.firestore.FieldValue.serverTimestamp()
    };
    await ref.set(newItem);
    res.status(201).json({ success: true, itemCode, reused: false, data: newItem });
  } catch (err) { next(err); }
});

// 품목 목록
router.get('/', async (req, res, next) => {
  try {
    const snap = await db.collection('items').orderBy('createdAt', 'desc').limit(100).get();
    res.json({ success: true, data: snap.docs.map(d => ({ id: d.id, ...d.data() })) });
  } catch (err) { next(err); }
});

// 특정 품목 + 연결된 도면 목록
router.get('/:itemCode', async (req, res, next) => {
  try {
    const [itemDoc, drawingsSnap] = await Promise.all([
      db.collection('items').doc(req.params.itemCode).get(),
      db.collection('drawings').where('itemCode', '==', req.params.itemCode).get()
    ]);
    if (!itemDoc.exists) return res.status(404).json({ success: false, error: '품목 없음' });
    res.json({
      success: true,
      data: { id: itemDoc.id, ...itemDoc.data() },
      drawings: drawingsSnap.docs.map(d => ({ id: d.id, ...d.data() }))
    });
  } catch (err) { next(err); }
});

module.exports = router;
```

## 3.7 Drawings API (DWG 업로드 + APS 변환 + 검토자 2명 승인)

```javascript
// src/routes/drawings.js
const express = require('express');
const router = express.Router();
const admin = require('firebase-admin');
const multer = require('multer');
const path = require('path');
const Joi = require('joi');
const { generateDwgNo } = require('../services/numbering');
const { getNextRevision } = require('../services/revision');
const { translateDwg } = require('../services/aps');

const db = admin.firestore();
const bucket = admin.storage().bucket();

// DWG 전용 업로드 (소용량 — 32MB 제한)
const upload = multer({
  storage: multer.memoryStorage(),
  limits: { fileSize: 32 * 1024 * 1024 },
  fileFilter: (req, file, cb) => {
    const ext = path.extname(file.originalname).toLowerCase();
    if (ext === '.dwg') return cb(null, true);
    cb(new Error('DWG 파일만 업로드할 수 있습니다.'));
  }
});

const uploadSchema = Joi.object({
  itemCode: Joi.string().required(),         // 기존/신규 품목코드 (중복 허용)
  dwgNo: Joi.string().optional(),            // 재업로드(Rev↑) 시 기존 dwgNo, 신규면 비움
  title: Joi.string().max(100).required(),
  changeReason: Joi.string().max(300).allow(''),
  authors: Joi.array().items(Joi.string()).min(1).required(),     // 담당자 여러 명
  reviewers: Joi.array().items(Joi.string()).length(2).required() // 검토자 정확히 2명
});

// 도면 업로드 (신규 도번 발급 또는 기존 도번 Revision↑)
router.post('/upload', upload.single('dwg'), async (req, res, next) => {
  try {
    if (!['designer', 'admin'].includes(req.user.role)) {
      return res.status(403).json({ success: false, error: '도면 등록 권한이 없습니다.' });
    }
    if (!req.file) return res.status(400).json({ success: false, error: 'DWG 파일이 필요합니다.' });

    // authors/reviewers는 JSON 문자열로 전달될 수 있으므로 파싱
    const body = {
      ...req.body,
      authors: typeof req.body.authors === 'string' ? JSON.parse(req.body.authors) : req.body.authors,
      reviewers: typeof req.body.reviewers === 'string' ? JSON.parse(req.body.reviewers) : req.body.reviewers
    };
    const { error, value } = uploadSchema.validate(body);
    if (error) return res.status(400).json({ success: false, error: error.details[0].message });

    // 품목 존재 확인
    const itemRef = db.collection('items').doc(value.itemCode);
    const itemDoc = await itemRef.get();
    if (!itemDoc.exists) {
      return res.status(404).json({ success: false, error: `품목 ${value.itemCode} 없음. 먼저 품목을 등록하세요.` });
    }

    // 도번/리비전 결정
    let dwgNo = value.dwgNo;
    let revision = 'A';
    if (dwgNo) {
      // 기존 도면 재업로드 → Released였으면 Rev↑
      const prev = await db.collection('drawings').doc(dwgNo).get();
      if (!prev.exists) return res.status(404).json({ success: false, error: `도번 ${dwgNo} 없음` });
      const prevData = prev.data();
      revision = prevData.status === 'Released' ? getNextRevision(prevData.revision) : prevData.revision;
    } else {
      dwgNo = await generateDwgNo(); // 신규 도번 발급
    }

    // Cloud Storage 저장
    const filePath = `drawings/${dwgNo}/REV_${revision}/${dwgNo}_REV_${revision}.dwg`;
    await bucket.file(filePath).save(req.file.buffer, {
      contentType: 'application/acad',
      metadata: { dwgNo, revision, itemCode: value.itemCode, uploadedBy: req.user.uid }
    });

    // APS 변환 잡 등록 → URN
    let apsUrn = null;
    try {
      apsUrn = await translateDwg(dwgNo, revision, req.file.buffer);
    } catch (e) {
      console.error(`[APS] 변환 등록 실패 (계속 진행): ${e.message}`);
    }

    // Firestore 저장 (도면 = In Work, 승인 대기 전 상태)
    const drawingData = {
      dwgNo,
      itemCode: value.itemCode,
      title: value.title,
      revision,
      status: 'In Work',
      filePath,
      fileUrl: `gs://${bucket.name}/${filePath}`,
      apsUrn,
      apsStatus: apsUrn ? 'inprogress' : 'failed',
      thumbnailUrl: null,                 // [미구현] 등각뷰 향후 확장용
      authors: value.authors,
      reviewers: value.reviewers,
      approvals: {},                      // 검토자별 승인 기록
      changeReason: value.changeReason || '',
      uploadedBy: req.user.uid,
      uploadedAt: admin.firestore.FieldValue.serverTimestamp()
    };

    const batch = db.batch();
    const drawingRef = db.collection('drawings').doc(dwgNo);
    // 헤드 문서: 현재 리비전 포인터
    batch.set(drawingRef, drawingData);
    // 리비전 스냅샷(불변 이력) — drawings/{dwgNo}/revisions/{rev}
    batch.set(drawingRef.collection('revisions').doc(revision), {
      revision,
      filePath,
      apsUrn,
      changeReason: value.changeReason || '',
      authors: value.authors,
      reviewers: value.reviewers,
      approvals: {},
      status: 'In Work',
      uploadedBy: req.user.uid,
      uploadedAt: admin.firestore.FieldValue.serverTimestamp()
    });
    batch.update(itemRef, {
      latestRev: revision,
      drawingCount: admin.firestore.FieldValue.increment(value.dwgNo ? 0 : 1)
    });
    await batch.commit();

    res.json({ success: true, dwgNo, itemCode: value.itemCode, revision, apsUrn });
  } catch (err) { next(err); }
});

// 담당자 제출 → In Review 전환
router.post('/:dwgNo/submit', async (req, res, next) => {
  try {
    const ref = db.collection('drawings').doc(req.params.dwgNo);
    const doc = await ref.get();
    if (!doc.exists) return res.status(404).json({ success: false, error: '도면 없음' });
    const data = doc.data();
    if (!data.authors.includes(req.user.uid) && req.user.role !== 'admin') {
      return res.status(403).json({ success: false, error: '담당자만 제출할 수 있습니다.' });
    }
    if (data.status !== 'In Work') {
      return res.status(400).json({ success: false, error: '작업 중 상태만 제출 가능합니다.' });
    }
    await ref.update({ status: 'In Review', submittedAt: admin.firestore.FieldValue.serverTimestamp() });
    res.json({ success: true, message: `${req.params.dwgNo} 검토 요청됨` });
  } catch (err) { next(err); }
});

// 검토자 승인 (2명 모두 승인해야 Released) ─ 핵심 로직
router.post('/:dwgNo/approve', async (req, res, next) => {
  try {
    const dwgNo = req.params.dwgNo;
    const ref = db.collection('drawings').doc(dwgNo);

    const result = await db.runTransaction(async (tx) => {
      const doc = await tx.get(ref);
      if (!doc.exists) throw Object.assign(new Error('도면 없음'), { status: 404 });
      const data = doc.data();

      // 호출자가 지정된 검토자인지 확인
      if (!data.reviewers.includes(req.user.uid)) {
        throw Object.assign(new Error('지정된 검토자가 아닙니다.'), { status: 403 });
      }
      if (data.status !== 'In Review') {
        throw Object.assign(new Error('검토 중 상태만 승인 가능합니다.'), { status: 400 });
      }

      // 본인 승인 기록
      const approvals = { ...data.approvals };
      approvals[req.user.uid] = { approved: true, at: new Date().toISOString() };

      // 검토자 2명 전원 승인 여부 확인
      const allApproved = data.reviewers.every(uid => approvals[uid]?.approved === true);

      if (!allApproved) {
        // 부분 승인 — 상태 유지(In Review)
        tx.update(ref, { approvals });
        return { released: false, approvedCount: Object.keys(approvals).length };
      }

      // 2명 모두 승인 → 헤드 문서 Released + 현재 리비전 스냅샷 Released
      tx.update(ref, {
        approvals,
        status: 'Released',
        releasedAt: admin.firestore.FieldValue.serverTimestamp()
      });
      tx.update(ref.collection('revisions').doc(data.revision), {
        approvals,
        status: 'Released',
        releasedAt: admin.firestore.FieldValue.serverTimestamp()
      });
      return { released: true, itemCode: data.itemCode, revision: data.revision };
    });

    // Released된 경우: 이 도번의 이전 리비전 스냅샷을 Obsolete로 전환
    if (result.released) {
      const revs = await db.collection('drawings').doc(dwgNo).collection('revisions')
        .where('status', '==', 'Released')
        .get();
      const batch = db.batch();
      revs.forEach(d => {
        if (d.id !== result.revision) batch.update(d.ref, { status: 'Obsolete' });
      });
      await batch.commit();
      // 품목 마스터 최신 리비전 동기화
      await db.collection('items').doc(result.itemCode).update({ latestRev: result.revision });
    }

    res.json({
      success: true,
      released: result.released,
      message: result.released
        ? `${dwgNo} 검토자 2명 전원 승인 완료 → Released`
        : `승인 기록됨 (${result.approvedCount}/2). 나머지 검토자 승인 대기 중`
    });
  } catch (err) {
    if (err.status) return res.status(err.status).json({ success: false, error: err.message });
    next(err);
  }
});

// 리비전 최근 내역 — 해당 dwgNo의 모든 리비전 스냅샷 (A,B,C...)
router.get('/:dwgNo/history', async (req, res, next) => {
  try {
    const headDoc = await db.collection('drawings').doc(req.params.dwgNo).get();

    const snap = await db.collection('drawings').doc(req.params.dwgNo)
      .collection('revisions')
      .orderBy('uploadedAt', 'desc')
      .get();
    res.json({
      success: true,
      head: { id: headDoc.id, ...headDoc.data() },
      revisions: snap.docs.map(d => ({ id: d.id, ...d.data() }))
    });
  } catch (err) { next(err); }
});

module.exports = router;
```

## 3.8 Search API (itemCode 또는 dwgNo → 도면 열람)

```javascript
// src/routes/search.js
const express = require('express');
const router = express.Router();
const admin = require('firebase-admin');
const { getTranslationStatus } = require('../services/aps');

const db = admin.firestore();

// 통합 검색: q가 DWG-로 시작하면 도번, 아니면 품목코드로 간주
router.get('/', async (req, res, next) => {
  try {
    const q = (req.query.q || '').trim().toUpperCase();
    if (!q) return res.json({ success: true, data: [] });

    let drawings = [];
    if (q.startsWith('DWG-')) {
      const doc = await db.collection('drawings').doc(q).get();
      if (doc.exists) drawings = [{ id: doc.id, ...doc.data() }];
    } else {
      // itemCode (부분 일치 prefix 검색)
      const snap = await db.collection('drawings')
        .where('itemCode', '>=', q)
        .where('itemCode', '<=', q + '')
        .get();
      drawings = snap.docs.map(d => ({ id: d.id, ...d.data() }));
    }

    // viewer는 Released만 노출
    if (req.user.role === 'viewer') {
      drawings = drawings.filter(d => d.status === 'Released');
    }
    res.json({ success: true, data: drawings });
  } catch (err) { next(err); }
});

// 도면 뷰어 정보 (APS URN + 변환 상태)
router.get('/:dwgNo/viewer', async (req, res, next) => {
  try {
    const doc = await db.collection('drawings').doc(req.params.dwgNo).get();

    const data = doc.data();
    if (!data.apsUrn) return res.json({ success: true, apsUrn: null, apsStatus: 'failed' });

    const status = await getTranslationStatus(data.apsUrn).catch(() => data.apsStatus);
    res.json({ success: true, apsUrn: data.apsUrn, apsStatus: status });
  } catch (err) { next(err); }
});

// APS Viewer가 사용할 2-legged 토큰 발급 (read-only)

    res.json({ success: true, access_token: cred.access_token, expires_in: cred.expires_in });

});
```

## 3.9 Admin API (관리자 1회 계정 발급)

```javascript
// src/routes/admin.js
const express = require('express');
const router = express.Router();
const admin = require('firebase-admin');
const Joi = require('joi');

const db = admin.firestore();

const userSchema = Joi.object({
  email: Joi.string().email().required(),
  name: Joi.string().min(2).max(50).required(),
  department: Joi.string().max(50).default(''),
  role: Joi.string().valid('admin', 'reviewer', 'designer', 'viewer').required()
});

// 관리자 전용 사용자 발급 (자가 가입 없음 — 이 API로만 계정 생성)
router.post('/users', async (req, res, next) => {
  try {
    if (req.user.role !== 'admin') {
      return res.status(403).json({ success: false, error: '관리자만 계정을 발급할 수 있습니다.' });
    }
    const { error, value } = userSchema.validate(req.body);
    if (error) return res.status(400).json({ success: false, error: error.details[0].message });

    // 임시 비밀번호로 Firebase Auth 사용자 생성
    const tempPassword = Math.random().toString(36).slice(2) + 'Aa1!';
    const userRecord = await admin.auth().createUser({
      email: value.email,
      password: tempPassword,
      displayName: value.name
    });

    await db.collection('users').doc(userRecord.uid).set({
      name: value.name,
      email: value.email,
      department: value.department,
      role: value.role,
      createdBy: req.user.uid,
      createdAt: admin.firestore.FieldValue.serverTimestamp()
    });

    // 비밀번호 설정 링크 생성 (관리자가 전달하거나 메일 발송)
    const resetLink = await admin.auth().generatePasswordResetLink(value.email);

    res.status(201).json({
      success: true,
      uid: userRecord.uid,
      email: value.email,
      passwordResetLink: resetLink,
      message: '계정 발급 완료. 비밀번호 설정 링크를 사용자에게 전달하세요.'
    });
  } catch (err) {
    if (err.code === 'auth/email-already-exists') {
      return res.status(409).json({ success: false, error: '이미 존재하는 이메일입니다.' });
    }
    next(err);
  }
});

// 사용자 목록 (관리자)
router.get('/users', async (req, res, next) => {
  try {
    if (req.user.role !== 'admin') {
      return res.status(403).json({ success: false, error: '권한 없음' });
    }
    const snap = await db.collection('users').orderBy('createdAt', 'desc').get();
    res.json({ success: true, data: snap.docs.map(d => ({ uid: d.id, ...d.data() })) });
  } catch (err) { next(err); }
});

module.exports = router;
```

## 3.10 인증 미들웨어

```javascript
// src/middleware/auth.js
const admin = require('firebase-admin');

async function verifyToken(req, res, next) {
  const authHeader = req.headers.authorization;
  if (!authHeader?.startsWith('Bearer ')) {
    return res.status(401).json({ success: false, error: '인증 토큰이 필요합니다.' });
  }
  try {
    const decoded = await admin.auth().verifyIdToken(authHeader.split('Bearer ')[1]);
    const userDoc = await admin.firestore().collection('users').doc(decoded.uid).get();
    if (!userDoc.exists) {
      // 관리자가 발급하지 않은 사용자(users 문서 없음)는 접근 거부
      return res.status(403).json({ success: false, error: '발급되지 않은 계정입니다.' });
    }
    req.user = {
      uid: decoded.uid,
      email: decoded.email,
      name: userDoc.data().name,
      role: userDoc.data().role
    };
    next();
  } catch (err) {
    res.status(401).json({ success: false, error: '유효하지 않은 토큰입니다.' });
  }
}

module.exports = { verifyToken };
```
