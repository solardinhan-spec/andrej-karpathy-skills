# 07. 보안 설계

## 7.1 역할 기반 접근 제어 (RBAC)

| 역할 | 설명 | 허용 작업 |
|------|------|-----------|
| `admin` | 시스템 관리자 | 모든 작업 + 사용자 관리 + 삭제 |
| `reviewer` | 설계 검토자 / 팀장 | 조회 + 승인 (Released 전환) |
| `designer` | 설계 엔지니어 | 조회 + Part/도면 등록 + BOM 편집 |
| `viewer` | 협력사 / 제조팀 | Released 도면 조회만 가능 |

## 7.2 Firestore Security Rules

```javascript
// firestore.rules
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ─────────────────────────────────────────────────────────
    // 역할 헬퍼 함수
    // ─────────────────────────────────────────────────────────

    function isAuthenticated() {
      return request.auth != null;
    }

    function getUserRole() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role;
    }

    function isAdmin() {
      return isAuthenticated() && getUserRole() == 'admin';
    }

    function isReviewer() {
      return isAuthenticated() && getUserRole() in ['admin', 'reviewer'];
    }

    function isDesigner() {
      return isAuthenticated() && getUserRole() in ['admin', 'reviewer', 'designer'];
    }

    function isAnyUser() {
      return isAuthenticated() && getUserRole() in ['admin', 'reviewer', 'designer', 'viewer'];
    }

    // ─────────────────────────────────────────────────────────
    // 품목 마스터 (parts)
    // ─────────────────────────────────────────────────────────
    match /parts/{partNo} {
      allow read: if isAnyUser();
      allow create: if isDesigner()
        && request.resource.data.keys().hasAll(['partNo', 'name', 'revision', 'status', 'createdBy'])
        && request.resource.data.status == 'In Work'; // 최초 등록 시 In Work만 허용
      allow update: if isDesigner()
        // Released → Obsolete 전환은 reviewer 이상만 가능
        && (request.resource.data.status != 'Obsolete' || isReviewer());
      allow delete: if false; // 품목 삭제 불가 (Obsolete 처리만 허용)
    }

    // ─────────────────────────────────────────────────────────
    // 도면 이력 (drawings) - 불변(Immutable) 설계
    // ─────────────────────────────────────────────────────────
    match /drawings/{drawingId} {
      allow read: if isAnyUser()
        // viewer는 Released 도면만 조회 가능
        && (getUserRole() != 'viewer' || resource.data.status == 'Released');
      allow create: if isDesigner();
      allow update: if
        // In Work → Released 전환: reviewer 이상
        (request.resource.data.status == 'Released' && isReviewer())
        // Released → Obsolete 전환: 시스템(admin)만
        || (request.resource.data.status == 'Obsolete' && isAdmin())
        // 기타 메타데이터 수정: designer
        || isDesigner();
      allow delete: if false; // 도면 삭제 영구 금지 (법적 보존 의무)
    }

    // ─────────────────────────────────────────────────────────
    // BOM 구조 (bom)
    // ─────────────────────────────────────────────────────────
    match /bom/{bomId} {
      allow read: if isAnyUser();
      allow write: if isDesigner();
      allow delete: if isAdmin();
    }

    // ─────────────────────────────────────────────────────────
    // 채번 시퀀스 (sequences) - 백엔드 서비스 계정만 접근
    // ─────────────────────────────────────────────────────────
    match /sequences/{seqId} {
      allow read, write: if false; // 클라이언트 직접 접근 완전 차단
    }

    // ─────────────────────────────────────────────────────────
    // ECN (ecn)
    // ─────────────────────────────────────────────────────────
    match /ecn/{ecnNo} {
      allow read: if isAnyUser();
      allow create: if isDesigner();
      allow update: if isReviewer();
      allow delete: if false;
    }

    // ─────────────────────────────────────────────────────────
    // 사용자 프로필 (users)
    // ─────────────────────────────────────────────────────────
    match /users/{userId} {
      // 본인 정보만 조회
      allow read: if isAuthenticated() && request.auth.uid == userId;
      // 관리자는 모든 사용자 조회 가능
      allow read: if isAdmin();
      // 본인 프로필 수정 (role 변경 불가)
      allow update: if isAuthenticated() && request.auth.uid == userId
        && !('role' in request.resource.data.diff(resource.data).affectedKeys());
      // 역할 변경은 admin만
      allow write: if isAdmin();
    }

    // ─────────────────────────────────────────────────────────
    // 감사 로그 (audit_logs) - 읽기 전용
    // ─────────────────────────────────────────────────────────
    match /audit_logs/{logId} {
      allow read: if isAdmin();
      allow write: if false; // 백엔드 서비스 계정만 쓰기
    }
  }
}
```

## 7.3 Cloud Storage 접근 규칙

```json
// storage.rules
rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {

    // 도면 파일 - 인증된 사용자만 접근
    match /drawings/{partNo}/{revision}/{fileName} {
      // 모든 인증 사용자 읽기 가능 (Signed URL로 단기 접근 제어)
      allow read: if request.auth != null;
      // 업로드는 백엔드 서비스 계정만 (서버 측에서만 처리)
      allow write: if false;
    }

    // 백업 파일
    match /backups/{allPaths=**} {
      allow read: if request.auth.token.role == 'admin';
      allow write: if false;
    }
  }
}
```

## 7.4 Secret Manager 활용

```bash
# 서비스 계정 키를 Secret Manager에 안전하게 보관
gcloud secrets create pdm-sa-key --replication-policy=automatic
gcloud secrets versions add pdm-sa-key --data-file=serviceAccountKey.json

# Drive OAuth2 토큰 저장
gcloud secrets create google-drive-oauth-token --replication-policy=automatic
echo -n "YOUR_OAUTH_TOKEN" | gcloud secrets versions add google-drive-oauth-token --data-file=-

# Cloud Run에서 Secret 접근 권한 부여
gcloud secrets add-iam-policy-binding pdm-sa-key \
  --member="serviceAccount:pdm-backend-sa@pdm-plm-semiconductor.iam.gserviceaccount.com" \
  --role="roles/secretmanager.secretAccessor"
```

```javascript
// Secret Manager에서 런타임 시 키 로드
const { SecretManagerServiceClient } = require('@google-cloud/secret-manager');

async function getSecret(secretName) {
  const client = new SecretManagerServiceClient();
  const [version] = await client.accessSecretVersion({
    name: `projects/${process.env.PROJECT_ID}/secrets/${secretName}/versions/latest`
  });
  return version.payload.data.toString('utf8');
}
```

## 7.5 감사 로그 (Audit Log) 미들웨어

```javascript
// src/middleware/auditLog.js
const admin = require('firebase-admin');

function auditLog(action) {
  return async (req, res, next) => {
    const originalJson = res.json.bind(res);

    res.json = async (body) => {
      if (body.success) {
        await admin.firestore().collection('audit_logs').add({
          action,
          userId: req.user?.uid,
          userName: req.user?.name || req.user?.email,
          role: req.user?.role,
          params: { ...req.params, ...req.body },
          ip: req.ip,
          userAgent: req.headers['user-agent'],
          result: 'success',
          timestamp: admin.firestore.FieldValue.serverTimestamp()
        }).catch(err => console.error('[AuditLog] 저장 실패:', err.message));
      }
      return originalJson(body);
    };

    next();
  };
}

module.exports = { auditLog };
```

```javascript
// 라우터에서 사용 예시
router.post('/approve', verifyToken, auditLog('DRAWING_APPROVE'), async (req, res) => { ... });
router.post('/upload', verifyToken, auditLog('DRAWING_UPLOAD'), upload.fields(...), async (req, res) => { ... });
```
